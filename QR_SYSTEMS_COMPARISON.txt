━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1️⃣  نظام QR القديم (للـ Sessions - تأكيد الحضور)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📍 Controller: QrCodeController
📍 Prefix: /qr

Routes:
  • /qr/session/{id}/image      → توليد QR للـ session
  • /qr/confirm/{token}          → تأكيد الحضور
  • /qr/session/{id}/reset       → reset الحضور
  • /qr/demo/{id}                → demo

الاستخدام:
  ✅ تأكيد حضور المريض للجلسة
  ✅ نظام gamification (نقاط + badges)
  ✅ مرتبط بـ Session entity

مثال:
  المريض يمسح QR → يأكد حضوره → يربح نقاط


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

2️⃣  نظام المشاعر الجديد (تتبع الحالة المزاجية)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📍 Controller: MoodController
📍 Prefix: /mood

Routes:
  • /mood/{patientId}            → اختيار المشاعر
  • /mood/save                   → حفظ المشاعر
  • /mood/dashboard              → Dashboard الإحصائيات
  • /mood/qr-codes               → توليد QR codes
  • /mood/test                   → صفحة اختبار

الاستخدام:
  ✅ تتبع المشاعر اليومية
  ✅ 3 خيارات: 😄 😐 😢
  ✅ مرتبط بـ MoodTracking entity

مثال:
  الطفل يمسح QR → يختار حالته → تتسجل في Dashboard
