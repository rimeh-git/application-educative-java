🎯 الهدف: المعالج يكتب feedback → الطفل يشوفه بعد scan QR


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
📝 الخطوة 1: المعالج يضيف Feedback
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. افتح Session:
   http://localhost:8000/session/1

2. انزل لقسم "💬 Feedback Thérapeute"

3. اكتب رسالة في الـ textarea:
   ┌─────────────────────────────────────────┐
   │ Bravo Lucas! Tu as très bien travaillé │
   │ aujourd'hui. Continue comme ça! 🌟     │
   └─────────────────────────────────────────┘

4. اضغط "✉️ Envoyer le feedback"

5. ✅ تظهر رسالة: "Feedback ajouté avec succès!"


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
📱 الخطوة 2: الطفل يمسح QR Code
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

🔹 الطريقة السريعة (Demo):
──────────────────────────────

1. في نفس صفحة Session
2. شوف قسم "QR Code Présence" (في الـ sidebar)
3. اضغط على زر "📱 Simuler le scan"
4. → يفتح صفحة جديدة (confirm)
5. → انزل للأسفل
6. → تشوف قسم "💬 Message du Thérapeute"

┌─────────────────────────────────────────────┐
│ 💬 Message du Thérapeute                   │
│ ┌─────────────────────────────────────────┐ │
│ │ 👨‍⚕️ Bravo Lucas! Tu as très bien      │ │
│ │    travaillé aujourd'hui. Continue     │ │
│ │    comme ça! 🌟                        │ │
│ │                                         │ │
│ │ 🕐 03/05/2026 à 23:45                  │ │
│ └─────────────────────────────────────────┘ │
└─────────────────────────────────────────────┘


🔹 الطريقة الحقيقية (Scan بالهاتف):
────────────────────────────────────────

1. افتح Session على الكمبيوتر
2. شوف QR Code في الـ sidebar
3. افتح كاميرا الهاتف
4. امسح QR Code
5. → يفتح رابط في المتصفح
6. → تشوف صفحة confirm مع الـ feedback


🔹 الطريقة المباشرة (بدون scan):
──────────────────────────────────

1. افتح Session
2. اضغط F12 (Developer Tools)
3. ابحث عن QR Code image
4. انسخ الـ URL من src attribute
5. مثال: http://localhost:8000/qr/confirm/abc123...
6. افتح الرابط في tab جديد
7. → تشوف الـ feedback

✏️ تعديل الـ Feedback
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

إذا تريد تعدل الـ feedback:

1. ارجع لصفحة Session
2. قسم "Feedback Thérapeute"
3. الـ textarea فيها الرسالة القديمة
4. عدلها
5. اضغط "🔄 Mettre à jour"
6. الطفل يمسح QR مرة ثانية → يشوف الرسالة الجديدة


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
🔍 استكشاف الأخطاء
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

❌ ما يظهر الـ feedback؟
→ تأكد أنك كتبت feedback وضغطت "Envoyer"
→ تأكد أنك تمسح QR للـ session الصحيح

❌ QR Code ما يخدم؟
→ استخدم زر "Simuler le scan" بدلاً من scan
→ أو انسخ الرابط يدوياً

❌ صفحة confirm فارغة؟
→ تأكد أن الـ session موجود
→ تأكد أن QR token موجود
